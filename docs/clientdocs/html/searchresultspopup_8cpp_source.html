79void SearchResultsPopup::showResults(const QJsonArray &users)
80{
81 // 1. Очищаем список от предыдущих результатов.
82 m_listWidget->clear();
83
84 // 2. Заполняем список новыми данными.
85 for (const QJsonValue &value : users) {
86 QJsonObject userObj = value.toObject();
87 QString displayName = userObj["displayname"].toString();
88 QString username = userObj["username"].toString();
89
90 // Создаем текстовое представление для отображения.
91 QListWidgetItem *item = new QListWidgetItem(displayName + " (@" + username + ")");
92 // Сохраняем `username` в `Qt::UserRole`. Это позволяет нам легко
93 // получить идентификатор пользователя при клике, не парся строку.
94 item->setData(Qt::UserRole, username);
95 m_listWidget->addItem(item);
96 }
97
98 // 3. Если результатов нет, просто скрываем окно и выходим.
99 int count = m_listWidget->count();
100 if (count == 0) {
101 hide();
102 return;
103 }
104
105 // --- 4. Динамический расчет высоты виджета ---
106 QFontMetrics fm(m_listWidget->font());
107 int itemHeight = fm.height() + 37; // Расчетная высота одного элемента (подобрана экспериментально).
108 const int maxVisibleItems = 4; // Максимальное количество элементов без прокрутки.
109
110 int targetHeight = 0;
111 if (count > maxVisibleItems) {
112 // Если элементов больше максимума, ограничиваем высоту и показываем скроллбар.
113 targetHeight = itemHeight * maxVisibleItems + m_listWidget->frameWidth() * 2;
114 m_listWidget->setVerticalScrollBarPolicy(Qt::ScrollBarAsNeeded);
115 } else {
116 // Если элементов меньше, делаем высоту точно по их количеству и скрываем скроллбар.
117 targetHeight = itemHeight * count + m_listWidget->frameWidth() * 2;
118 m_listWidget->setVerticalScrollBarPolicy(Qt::ScrollBarAlwaysOff);
119 }
120
121 // Устанавливаем рассчитанную высоту для всего виджета.
122 this->setFixedHeight(targetHeight);
123
124 // 5. Показываем виджет, если он был скрыт.
125 if (!isVisible()) {
126 show();
127 }
