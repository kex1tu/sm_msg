61void ChatMessageDelegate::paint(QPainter *painter, const QStyleOptionViewItem &option, const QModelIndex &index) const
62{
63 // --- 1. Подготовка ---
64 // Сбрасываем стандартное выделение и наведение, чтобы управлять ими вручную (hover-эффект пузыря).
65 QStyleOptionViewItem opt = option;
66 opt.state &= ~(QStyle::State_Selected | QStyle::State_MouseOver | QStyle::State_HasFocus);
67 QStyledItemDelegate::paint(painter, opt, index); // Рисуем фон и другие базовые вещи.
68
69 painter->save(); // Сохраняем текущее состояние painter'а (трансформации, кисти, перья).
70 painter->setRenderHint(QPainter::Antialiasing); // Включаем сглаживание для красивых фигур.
71
72
73 // Получаем объект сообщения из модели.
74 ChatMessage message = index.data(Qt::UserRole).value<ChatMessage>();
75 QRect originalRect = option.rect;
76
77 // Определяем базовые константы геометрии.
78 int margin = 10; // Внешний отступ пузыря от краев элемента.
79 int padding = 10; // Внутренний отступ от краев пузыря до контента.
80 int borderRadius = 15; // Радиус скругления углов пузыря.
81 int verticalSpacing = 10; // Вертикальный отступ между сообщениями.
82 QFontMetrics fm(painter->font());
83
84
85 // Уменьшаем рабочую область, чтобы учесть вертикальный отступ.
86 QRect contentRect = originalRect.adjusted(0, verticalSpacing / 2, 0, -verticalSpacing / 2);
87
88 // --- 2. Отрисовка цитаты (ответа) ---
89 ChatMessage repliedMsg;
90 int quoteHeight = 0;
91 bool foundReply = false;
92 if (message.replyToId > 0) {
93 // Запрашиваем данные цитируемого сообщения у модели по его ID.
94 foundReply = m_model->getMessageById(message.replyToId, repliedMsg);
95 }
96
97
98 if (foundReply) {
99 quoteHeight = fm.height() * 2 + 15; // Расчетная высота блока цитаты.
100
101 // Рассчитываем геометрию блока цитаты.
102 int nameWidth = fm.horizontalAdvance(repliedMsg.fromUser);
103 QString elidedText = fm.elidedText(repliedMsg.payload, Qt::ElideRight, 250); // Обрезаем длинный текст.
104 int textWidth = fm.horizontalAdvance(elidedText);
105 int quoteContentWidth = std::max(nameWidth, textWidth);
106 int quoteTotalWidth = quoteContentWidth + padding + 4 + 8; // Ширина = контент + отступы + полоса.
107
108 QRectF quoteRect(0, 0, quoteTotalWidth, quoteHeight - 5);
109
110 // Выравниваем блок цитаты по правому или левому краю.
111 if (message.isOutgoing) {
112 quoteRect.moveTopRight(contentRect.topRight() - QPoint(margin, 0));
113 } else {
114 quoteRect.moveTopLeft(contentRect.topLeft() + QPoint(margin, 0));
115 }
116
117 // Рисуем цветную вертикальную полосу слева от цитаты.
118 QRectF colorBarRect = quoteRect.adjusted(padding, 4, 0, -4);
119 colorBarRect.setWidth(4);
120 painter->setBrush(QColor("#E072A4"));
121 painter->setPen(Qt::NoPen);
122 painter->drawRoundedRect(colorBarRect, 2, 2);
123
124 // Рисуем имя автора и текст цитаты.
125 QRectF quoteTextRect = quoteRect.adjusted(colorBarRect.width() + padding + 4, 5, -padding, -5);
126 painter->setPen(QColor("#E072A4"));
127 painter->drawText(quoteTextRect, Qt::AlignTop | Qt::AlignLeft, repliedMsg.fromUser);
128 painter->setPen(Qt::gray);
129 QRectF repliedTextRect = quoteTextRect.adjusted(0, fm.height(), 0, 0);
130 painter->drawText(repliedTextRect, Qt::AlignTop | Qt::AlignLeft, elidedText);
131
132 // Сдвигаем область для отрисовки основного сообщения вниз.
133 contentRect.setTop(contentRect.top() + quoteHeight);
134 }
135
136
137 // --- 3. Расчет геометрии основного пузыря сообщения ---
138 // Ограничиваем максимальную ширину пузыря (75% от ширины виджета).
139 int textWidth = contentRect.width() * 0.75 - 2 * padding;
140 if (textWidth <= 0) textWidth = 300; // Минимальная ширина на случай ошибок.
141
142 // Рассчитываем прямоугольник, необходимый для отрисовки текста с переносами.
143 QRect payloadRect = fm.boundingRect(QRect(0, 0, textWidth, 0), Qt::TextWrapAnywhere, message.payload);
144 int metaDataHeight = fm.height(); // Высота для строки с временем и статусом.
145
146 // Формируем строку метаданных.
147 QString metaText;
148 if (message.isEdited) metaText += "(изм.) ";
149 metaText += message.timestamp.mid(11, 5); // "HH:mm"
150 int metaTextWidth = fm.horizontalAdvance(metaText);
151
152 // Ширина пузыря - это максимум из ширины текста и ширины метаданных.
153 int bubbleContentWidth = std::max(payloadRect.width(), metaTextWidth);
154 int bubbleContentHeight = payloadRect.height() + metaDataHeight + 4;
155
156 QRect bubbleRect(0, 0, bubbleContentWidth + 2 * padding, bubbleContentHeight + 2 * padding);
157
158 // Устанавливаем минимальную ширину, чтобы короткие сообщения (типа "ок") не выглядели слишком узкими.
159 int minBubbleWidth = 100;
160 if (bubbleRect.width() < minBubbleWidth) bubbleRect.setWidth(minBubbleWidth);
161
162 // Выравниваем пузырь по правому (исходящие) или левому (входящие) краю.
163 if (message.isOutgoing) bubbleRect.moveTopRight(contentRect.topRight() - QPoint(margin, 0));
164 else bubbleRect.moveTopLeft(contentRect.topLeft() + QPoint(margin, 0));
165
166 // --- 4. Отрисовка основного пузыря ---
167 QColor bubbleColor = message.isOutgoing ? QColor("#E072A4").darker(150) : QColor("#3D383A");
168 painter->setBrush(bubbleColor);
169 painter->setPen(Qt::NoPen);
170 painter->drawRoundedRect(bubbleRect, borderRadius, borderRadius);
171
172 // Рисуем текст сообщения.
173 QRect textDrawRect = bubbleRect.adjusted(padding, padding, -padding, -padding);
174 textDrawRect.setHeight(payloadRect.height());
175 painter->setPen(Qt::white);
176 QTextOption textOption(Qt::AlignLeft | Qt::AlignVCenter);
177 textOption.setWrapMode(QTextOption::WrapAnywhere); // Включаем перенос слов.
178 painter->drawText(textDrawRect, message.payload, textOption);
179
180 // --- 5. Отрисовка метаданных (время и статус) ---
181 QRect baseMetaRect = bubbleRect.adjusted(padding, padding, -padding, -padding);
182 baseMetaRect.setTop(textDrawRect.bottom() + 4);
183 baseMetaRect.setHeight(metaDataHeight);
184
185 if (message.isOutgoing) {
186 // Для исходящих сообщений рисуем и текст, и иконку статуса.
187 int iconSize = fm.height() - 2;
188 int iconPadding = 3;
189
190 // Рассчитываем место для иконки справа.
191 QRect iconRect(baseMetaRect.right() - iconSize, baseMetaRect.top() + (baseMetaRect.height() - iconSize) / 2, iconSize, iconSize);
192 // Оставшееся место слева от иконки - для текста.
193 QRect textMetaRect = baseMetaRect;
194 textMetaRect.setRight(iconRect.left() - iconPadding);
195
196 // Рисуем текст (время, "изм.").
197 QPen textPen = (message.status == ChatMessage::Read) ? QColor(70, 150, 255) : Qt::gray;
198 painter->setPen(textPen);
199 painter->drawText(textMetaRect, Qt::AlignRight | Qt::AlignVCenter, metaText);
200
201 // --- Продвинутая отрисовка и перекрашивание SVG иконки ---
202 ChatMessage::MessageStatus statusToRender = message.status;
203 if (statusToRender == ChatMessage::Read) statusToRender = ChatMessage::Delivered; // Для "прочитано" используем иконку "доставлено".
204
205 QSvgRenderer* renderer = m_statusRenderers.value(statusToRender, nullptr);
206 if (renderer && renderer->isValid()) {
207 QPixmap pixmap(iconRect.size());
208 pixmap.fill(Qt::transparent);
209
210 // 1. Рендерим SVG на временный QPixmap.
211 QPainter pixmapPainter(&pixmap);
212 renderer->render(&pixmapPainter);
213 pixmapPainter.end();
214
215 // 2. "Заливаем" отрендеренное изображение нужным цветом.
216 QPainter effectPainter(&pixmap);
217 effectPainter.setCompositionMode(QPainter::CompositionMode_SourceIn);
218 QColor iconColor = (message.status == ChatMessage::Read) ? QColor(70, 150, 255) : Qt::gray;
219 effectPainter.fillRect(pixmap.rect(), iconColor);
220 effectPainter.end();
221
222 // 3. Рисуем итоговый перекрашенный QPixmap на основной холст.
223 painter->drawPixmap(iconRect, pixmap);
224 }
225 } else {
226 // Для входящих сообщений просто рисуем текст справа.
227 painter->setPen(Qt::gray);
228 painter->drawText(baseMetaRect, Qt::AlignRight | Qt::AlignVCenter, metaText);
229
230
231 }
232
233 painter->restore(); // Восстанавливаем состояние painter'а.
234}

245QSize ChatMessageDelegate::sizeHint(const QStyleOptionViewItem &option, const QModelIndex &index) const
246{
247 ChatMessage message = index.data(Qt::UserRole).value<ChatMessage>();
248 int padding = 10;
249 int verticalSpacing = 10;
250 QFontMetrics fm(option.font);
251
252 // 1. Рассчитываем высоту блока цитаты, если он есть.
253 int quoteHeight = 0;
254 if (message.replyToId > 0) {
255 ChatMessage repliedMsg;
256 if (m_model->getMessageById(message.replyToId, repliedMsg)) {
257 quoteHeight = fm.height() * 2 + 15;
258 }
259 }
260
261 // 2. Рассчитываем высоту основного текста с учетом переносов.
262 int textWidth = option.rect.width() * 0.75 - 2 * padding;
263 if (textWidth <= 0) textWidth = 300;
264 QRect payloadRect = fm.boundingRect(QRect(0, 0, textWidth, 0), Qt::TextWrapAnywhere, message.payload);
265
266 // 3. Высота строки метаданных.
267 int metaDataHeight = fm.height();
268
269 // 4. Суммарная высота пузыря.
270 int bubbleHeight = payloadRect.height() + metaDataHeight + 4 + (2 * padding);
271
272 // 5. Итоговая высота всего элемента.
273 int totalHeight = quoteHeight + bubbleHeight + verticalSpacing;
274
275 return QSize(option.rect.width(), totalHeight);
