124bool ChatViewWidget::eventFilter(QObject *watched, QEvent *event)
125{
126 // Проверяем, что событие пришло от нашего поля ввода и что это нажатие клавиши.
127 if (watched == ui->messageTextEdit && event->type() == QEvent::KeyPress) {
128 QKeyEvent *keyEvent = static_cast<QKeyEvent*>(event);
129
130 // Если нажат Enter или Return...
131 if (keyEvent->key() == Qt::Key_Return || keyEvent->key() == Qt::Key_Enter) {
132 // ...и при этом НЕ зажата клавиша Shift...
133 if (!(keyEvent->modifiers() & Qt::ShiftModifier)) {
134 ui->sendButton->click(); // ...симулируем клик по кнопке "Отправить".
135 return true; // Поглощаем событие, чтобы стандартный обработчик QTextEdit не вставил символ новой строки.
136 }
137 // Если Shift зажат, `return true` не сработает, и QTextEdit обработает событие, вставив перенос строки.
138 }
139 }
140
141 // Для всех остальных событий вызываем обработчик базового класса.
142 return QWidget::eventFilter(watched, event);
143}

244void ChatViewWidget::onChatContextMenuRequested(const QPoint &pos)
245{
246 QModelIndex index = ui->chatHistoryView->indexAt(pos);
247 if (!index.isValid()) return;
248
249 ChatMessage msg = index.data(Qt::UserRole).value<ChatMessage>();
250 QMenu contextMenu(this);
251 QAction *replyAction = contextMenu.addAction("Ответить");
252 QAction *editAction = contextMenu.addAction("Редактировать");
253 QAction *deleteAction = contextMenu.addAction("Удалить");
254
255 // Разрешаем/запрещаем действия в зависимости от того, наше ли это сообщение.
256 if (!msg.isOutgoing) {
257 editAction->setEnabled(false);
258 deleteAction->setEnabled(false);
259 }
260
261 // Показываем меню и ждем выбора пользователя.
262 QAction *selectedAction = contextMenu.exec(ui->chatHistoryView->viewport()->mapToGlobal(pos));
263
264 // Обрабатываем выбор пользователя, эмитируя соответствующие сигналы.
265 if (selectedAction == replyAction) {
266 onMessageDoubleClicked(index); // Используем ту же логику, что и для двойного клика.
267 } else if (selectedAction == editAction) {
268 emit editMessageRequested(msg.id, msg.payload);
269 } else if (selectedAction == deleteAction) {
270 emit deleteMessageRequested(msg.id);
271 }
272}

319QString ChatViewWidget::formatLastSeen(const User &user)
320{
321 if (user.isOnline) {
322 return "в сети";
323 }
324 if (user.lastSeen.isEmpty()) {
325 return "не в сети";
326 }
327 QDateTime lastSeenTime = QDateTime::fromString(user.lastSeen, Qt::ISODate);
328 if (!lastSeenTime.isValid()) {
329 return "не в сети";
330 }
331 QDateTime now = QDateTime::currentDateTime();
332 qint64 diffSeconds = lastSeenTime.secsTo(now);
333
334 if (diffSeconds < 60) {
335 return "был(а) только что";
336 } else if (diffSeconds < 3600) {
337 int minutes = diffSeconds / 60;
338 return QString("был(а) %1 %2 назад").arg(minutes).arg(pluralize(minutes, "минуту", "минуты", "минут"));
339 } else if (lastSeenTime.date() == now.date()) {
340 return "был(а) сегодня в " + lastSeenTime.toString("HH:mm");
341 } else if (lastSeenTime.date() == now.date().addDays(-1)) {
342 return "был(а) вчера в " + lastSeenTime.toString("HH:mm");
343 } else {
344 return "был(а) " + QLocale::system().toString(lastSeenTime, QLocale::ShortFormat);
345 }
346}
