85void LoginWidget::onTogglePasswordVisibilityTriggered()
86{
87 // Проверяем текущий режим отображения.
88 if (ui->loginPasswordEdit->echoMode() == QLineEdit::Password) {
89 // Если пароль был скрыт -> показываем его.
90 ui->loginPasswordEdit->setEchoMode(QLineEdit::Normal);
91 // Меняем иконку на "открытый глаз".
92 m_passwordVisibilityAction->setIcon(QIcon(":/icons/eye.png"));
93 } else {
94 // Если пароль был виден -> скрываем его.
95 ui->loginPasswordEdit->setEchoMode(QLineEdit::Password);
96 // Меняем иконку обратно на "закрытый глаз".
97 m_passwordVisibilityAction->setIcon(QIcon(":/icons/crossed_eye.png"));
98 }
