46QVariant ContactListModel::data(const QModelIndex &index, int role) const
47{
48 // Проверка на валидность индекса и наличие указателя на MainWindow.
49 if (!index.isValid() || !m_mainWindow) return QVariant();
50
51 // Получаем username, который является нашим ключом/идентификатором.
52 QString username = m_contactUsernames.at(index.row());
53
54 // Получаем полный объект User из кэша MainWindow.
55 const User user = m_mainWindow->getUserFromCache(username);
56
57 // В зависимости от роли, возвращаем соответствующее поле из кэшей MainWindow.
58 // Это и есть суть паттерна "модель-адаптер".
59 switch (role) {
60 case Qt::DisplayRole:
61 return user.displayName; // Текст, который будет отображаться по умолчанию.
62 case UsernameRole:
63 return user.username;
64 case IsOnlineRole:
65 return user.isOnline;
66 case IsTypingRole:
67 // Дополнительная логика: не показывать статус "печатает" для чата, который и так открыт.
68 return (user.isTyping && username != m_mainWindow->getCurrentChatPartner().username);
69 case LastMessageRole:
70 // Обращаемся к кэшу истории чатов.
71 if (m_mainWindow->getChatCache().contains(username)) {
72 const auto& messages = m_mainWindow->getChatCache().value(username).messages;
73 if (!messages.isEmpty()) {
74 return messages.last().payload; // Возвращаем текст последнего сообщения.
75 }
76 }
77 return QVariant(); // Возвращаем пустой QVariant, если истории нет.
78 case UnreadCountRole:
79 return m_mainWindow->getUnreadCounts().value(username, 0); // 0, если для юзера нет записей.
80 }
81
82 return QVariant();
83}

125void ContactListModel::updateContacts(const QStringList &newUsernames)
126{
127 const QStringList oldUsernames = m_contactUsernames;
128
129 // Используем QSet для сверхбыстрой проверки наличия элемента (O(1) в среднем).
130 const QSet<QString> newUsernamesSet = QSet<QString>(newUsernames.begin(), newUsernames.end());
131 const QSet<QString> oldUsernamesSet = QSet<QString>(oldUsernames.begin(), oldUsernames.end());
132
133 // --- Фаза 1: Удаление старых контактов ---
134 // Итерируемся в обратном порядке (с конца к началу), чтобы индексы
135 // не "съезжали" при удалении элементов из списка.
136 for (int i = oldUsernames.count() - 1; i >= 0; --i) {
137 if (!newUsernamesSet.contains(oldUsernames[i])) {
138 // Если старого юзера нет в новом списке, удаляем его.
139 beginRemoveRows(QModelIndex(), i, i); // Уведомляем view, что строка 'i' будет удалена.
140 m_contactUsernames.removeAt(i);
141 endRemoveRows(); // Завершаем операцию удаления.
142 }
143 }
144
145 // --- Фаза 2: Добавление новых контактов ---
146 for (int i = 0; i < newUsernames.count(); ++i) {
147 if (!oldUsernamesSet.contains(newUsernames[i])) {
148 // Если новый юзер не был в старом списке, добавляем его.
149 // Так как newUsernames уже отсортирован, мы просто вставляем его
150 // на ту же позицию 'i'.
151 beginInsertRows(QModelIndex(), i, i); // Уведомляем view, что в позицию 'i' будет вставлена строка.
152 m_contactUsernames.insert(i, newUsernames[i]);
153 endInsertRows(); // Завершаем операцию вставки.
154 }
155 }
156 // Примечание: Эта реализация не обрабатывает "перемещение" элементов,
157 // а рассматривает его как удаление старого и вставку нового. Для простого
158 // списка контактов этого более чем достаточно.
159}
