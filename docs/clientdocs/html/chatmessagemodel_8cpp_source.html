44QVariant ChatMessageModel::data(const QModelIndex &index, int role) const
45{
46 // Проверка на валидность индекса, чтобы избежать выхода за пределы массива.
47 if (!index.isValid() || index.row() >= m_messages.count())
48 return QVariant();
49
50 // Получаем константную ссылку на объект сообщения по индексу.
51 const ChatMessage &message = m_messages.at(index.row());
52
53 // Наша основная логика: по роли Qt::UserRole мы возвращаем весь объект ChatMessage.
54 // Делегат затем извлечет этот объект и возьмет из него все необходимые поля для отрисовки.
55 if (role == Qt::UserRole) {
56 return QVariant::fromValue(message);
57 }
58
59 // Для других стандартных ролей (DisplayRole, DecorationRole и т.д.) мы не возвращаем ничего.
60 return QVariant();
61}

137void ChatMessageModel::prependMessages(const QList<ChatMessage> &messages)
138{
139 if (messages.isEmpty()) return;
140
141 beginInsertRows(QModelIndex(), 0, messages.count() - 1);
142
143 // Итерируемся в обратном порядке, чтобы сохранить правильный порядок сообщений при вставке.
144 for (int i = messages.count() - 1; i >= 0; --i) {
145 const auto& msg = messages.at(i);
146 m_messages.prepend(msg);
147 m_messageMap[msg.id] = msg;
148 }
149
150 endInsertRows();
