268void MainWindow::processVisibleMessages()
269{
270 qDebug() <<"processVisibleMessages";
271 QListView* view = m_chatViewWidget->chatHistoryView();
272
273 // Проверка, что чат активен и у него есть модель.
274 if (!view->model() || m_currentChatPartner.username.isEmpty()) {
275 return;
276 }
277
278 // Получаем индексы первого и последнего видимого элемента в области просмотра.
279 QModelIndex firstVisible = view->indexAt(view->rect().topLeft());
280 QModelIndex lastVisible = view->indexAt(view->rect().bottomLeft());
281
282 // Если нет видимых элементов (например, чат пуст), выходим.
283 if (!firstVisible.isValid()) {
284 return;
285 }
286
287 // Проходимся в цикле от первого видимого до последнего.
288 for (int row = firstVisible.row(); row <= lastVisible.row(); ++row) {
289 QModelIndex index = m_chatModel->index(row, 0);
290 if (!index.isValid()) continue; // Пропускаем, если индекс невалидный.
291
292 // Получаем объект сообщения из модели.
293 ChatMessage msg = index.data(Qt::UserRole).value<ChatMessage>();
294
295 // Главное условие: если сообщение ВХОДЯЩЕЕ и его статус "Доставлено"...
296 if (!msg.isOutgoing && (msg.status == ChatMessage::Delivered|| msg.status == ChatMessage::Sent)) {
297 // ...то мы эмитируем сигнал от модели.
298 // Модель сама изменит статус на "Прочитано" и уведомит MainWindow,
299 // что нужно отправить сетевой запрос.
300 // Мы не вызываем markMessageAsRead напрямую, чтобы сохранить инкапсуляцию,
301 // хотя это тоже был бы рабочий вариант. Эмитирование сигнала здесь более гибко.
302 emit m_chatModel->messageNeedsReadReceipt(msg.id);
303 }
304 }

525void MainWindow::sendJson(const QJsonObject& json)
526{
527 qDebug() << "---------------------------------";
528 qDebug() << "[CLIENT SENDING] Preparing to send JSON of type:" << json["type"].toString();
529 qDebug() << "[CLIENT SENDING] Full JSON content:" << json;
530 qDebug() << "---------------------------------";
531
532 // 1. Преобразуем JSON-объект в байтовый массив.
533 QByteArray jsonData = QJsonDocument(json).toJson(QJsonDocument::Compact);
534
535 // 2. Создаем блок для отправки.
536 QByteArray block;
537 QDataStream out(&block, QIODevice::WriteOnly);
538 out.setVersion(QDataStream::Qt_6_2); // Важно для совместимости с сервером.
539
540 // 3. Записываем в начало блока "пустое" место (0) для размера.
541 out << (quint32)0;
542
543 // 4. Записываем сами данные.
544 out << jsonData;
545
546 // 5. Возвращаемся в начало блока.
547 out.device()->seek(0);
548
549 // 6. Записываем в "пустое" место реальный размер всего блока минус размер самого поля для размера.
550 out << (quint32)(block.size() - sizeof(quint32));
551
552 // 7. Отправляем итоговый блок по сокету.
553 socket->write(block);
554}

565void MainWindow::onReadyRead() {
566 QDataStream in(socket);
567 in.setVersion(QDataStream::Qt_6_2); // Версия должна совпадать с сервером.
568
569 // Запускаем бесконечный цикл, который будет читать данные, пока они есть в буфере.
570 while(true) {
571 // Если мы не знаем размер следующего блока...
572 if (m_nextBlockSize == 0) {
573 // ...проверяем, достаточно ли данных в буфере, чтобы прочитать сам размер (4 байта).
574 if (socket->bytesAvailable() < sizeof(quint32)) {
575 break; // Данных мало, выходим и ждем следующего вызова onReadyRead.
576 }
577 // Читаем размер следующего блока.
578 in >> m_nextBlockSize;
579 }
580
581 // Теперь мы знаем размер. Проверяем, достаточно ли данных в буфере,
582 // чтобы прочитать весь блок целиком.
583 if (socket->bytesAvailable() < m_nextBlockSize) {
584 break; // Пакет еще не пришел целиком. Выходим и ждем.
585 }
586
587 // Данных достаточно, читаем сам JSON-пакет.
588 QByteArray jsonData;
589 in >> jsonData;
590
591 // Сбрасываем размер блока, чтобы быть готовыми к чтению следующего.
592 m_nextBlockSize = 0;
593
594 // Парсим JSON.
595 QJsonDocument doc = QJsonDocument::fromJson(jsonData);
596 if (doc.isNull() || !doc.isObject()) {
597 qDebug() << "[CLIENT] Failed to parse JSON or it's not an object.";
598 continue; // Пропускаем поврежденный пакет.
599 }
600
601 // --- Маршрутизация команды ---
602 QJsonObject response = doc.object();
603 QString type = response["type"].toString();
604 qDebug() << "[CLIENT] Processing message of type" << type;
605
606 // Ищем обработчик для данного типа команды в нашей карте.
607 if (m_responseHandlers.contains(type)) {
608 // Если найден, вызываем соответствующий метод-обработчик.
609 ResponseHandler handler = m_responseHandlers[type];
610 (this->*handler)(response);
611 } else {
612 qDebug() << "[CLIENT] No handler found for message type:" << type;
613 }
614 }

834void MainWindow::handleHistoryData(const QJsonObject& response)
835{
836 // 1. Проверяем, что полученная история относится к чату, который открыт в данный момент.
837 // Это предотвращает редкую ситуацию, когда пользователь очень быстро переключает чаты,
838 // а ответ от сервера на старый запрос приходит с опозданием.
839 QString historyForUser = response["with_user"].toString();
840 if (historyForUser != m_currentChatPartner.username) return;
841
842 // --- 2. Парсинг и обработка сообщений ---
843 QJsonArray history = response["history"].toArray();
844 qDebug() << "[CLIENT] Displaying initial" << history.count() << "messages for" << historyForUser;
845
846 QList<ChatMessage> messages; // Временный список для хранения распарсенных сообщений.
847 for (const QJsonValue &value : history) {
848 QJsonObject msgObj = value.toObject();
849 ChatMessage msg;
850
851 // Заполняем структуру ChatMessage данными из JSON.
852 msg.id = msgObj["id"].toDouble();
853 msg.fromUser = msgObj["fromUser"].toString();
854 msg.toUser = msgObj["toUser"].toString();
855 msg.payload = msgObj["payload"].toString();
856 msg.timestamp = msgObj["timestamp"].toString();
857 msg.replyToId = msgObj["reply_to_id"].toDouble();
858 msg.isEdited = msgObj["is_edited"].toInt();
859 msg.isOutgoing = (msg.fromUser == m_currentUsername); // Флаг, который мы вычисляем на клиенте.
860
861 // Определяем статус сообщения (для исходящих).
862 if(msgObj["is_read"].toInt() == 1) {
863 msg.status = ChatMessage::Read;
864 } else if (msgObj["is_delivered"].toInt() == 1) {
865 msg.status = ChatMessage::Delivered;
866 } else {
867 msg.status = ChatMessage::Sent;
868 }
869
870 messages.append(msg);
871 }
872 ChatCache& cache = m_chatHistoryCache[historyForUser];
873 cache.messages = messages;
874 if (!messages.isEmpty()) {
875 cache.oldestMessageId = messages.first().id;
876 } else {
877 cache.allMessagesLoaded = true;
878 }
879 m_oldestMessageId = cache.oldestMessageId; // Обновляем состояние MainWindow
880
881 // Просто и чисто: очищаем модель и загружаем в нее новые данные.
882 m_chatModel->clearMessages();
883 m_chatModel->addMessages(messages);
884
885 // --- 4. Обновление UI и состояния ---
886 QMetaObject::invokeMethod(m_chatViewWidget->chatHistoryView(), "scrollToBottom", Qt::QueuedConnection);
887
888 QTimer::singleShot(50, this, &MainWindow::processVisibleMessages);
889 m_isLoadingHistory = false;
890}

901void MainWindow::handlePrivateMessage(const QJsonObject& response) {
902
903 // --- Ветка 1: Обработка "echo"-ответа на наше собственное сообщение ---
904 // Сервер, получив наше сообщение, сохраняет его в БД, присваивает ему ID
905 // и возвращает нам его обратно с этим ID и `temp_id`, который мы ему прислали.
906 QString tempId = response["temp_id"].toString();
907 if (!tempId.isEmpty()) {
908 qDebug() << "[CLIENT] Received ECHO for temp_id:" << tempId;
909
910 // 1. Парсим подтвержденное сообщение.
911 ChatMessage msg;
912 msg.id = response["id"].toDouble(); // <-- Самое важное: получаем реальный ID.
913 msg.tempId = tempId;
914 msg.fromUser = response["fromUser"].toString();
915 msg.toUser = response["toUser"].toString();
916 msg.payload = response["payload"].toString();
917 msg.timestamp = response["timestamp"].toString(); // Серверное время.
918 msg.replyToId = response["reply_to_id"].toDouble();
919 msg.isOutgoing = true;
920 msg.status = ChatMessage::MessageStatus::Sent; // Статус меняется с "Sending" на "Sent".
921 msg.isEdited = false;
922
923 // 2. Обновляем сообщение в модели, видимой на экране.
924 // Модель найдет сообщение по `tempId` и заменит его на `msg`.
925 m_chatModel->confirmMessage(tempId, msg);
926
927 // 3. Обновляем сообщение в глобальном кэше истории.
928 QString chatPartner = msg.toUser;
929 if (m_chatHistoryCache.contains(chatPartner)) {
930 QList<ChatMessage>& messagesInCache = m_chatHistoryCache[chatPartner].messages;
931 // Ищем временное сообщение в кэше и заменяем его на подтвержденное.
932 for (int i = 0; i < messagesInCache.size(); ++i) {
933 if (messagesInCache[i].tempId == tempId) {
934 messagesInCache[i] = msg;
935 qDebug() << "[CACHE] Confirmed message with temp_id:" << tempId << "in cache for" << chatPartner;
936 break;
937 }
938 }
939 }
940
941 return; // Завершаем выполнение, так как это был echo-ответ.
942 }
943
944 // --- Ветка 2: Обработка нового входящего сообщения от другого пользователя ---
945 // Если `temp_id` пустой, значит, это сообщение от кого-то другого.
946
947 // 1. Парсим входящее сообщение.
948 ChatMessage incomingMsg;
949 incomingMsg.id = response["id"].toDouble();
950 incomingMsg.fromUser = response["fromUser"].toString();
951 incomingMsg.toUser = response["toUser"].toString();
952 incomingMsg.payload = response["payload"].toString();
953 incomingMsg.timestamp = response["timestamp"].toString();
954 incomingMsg.replyToId = response["reply_to_id"].toDouble();
955 incomingMsg.isOutgoing = false;
956 incomingMsg.isEdited = false;
957
958 // Определяем статус, присланный сервером.
959 if (response["is_read"].toInt() == 1) {
960 incomingMsg.status = ChatMessage::Read;
961 } else if (response["is_delivered"].toInt() == 1) {
962 incomingMsg.status = ChatMessage::Delivered;
963 } else {
964 incomingMsg.status = ChatMessage::Sent;
965 }
966
967 // 2. Обновляем UI списка контактов, чтобы показать превью этого нового сообщения.
968 updateContactItem(incomingMsg.fromUser);
969
970 // 3. Добавляем сообщение в кэш соответствующего чата.
971 QString chatPartner = incomingMsg.fromUser;
972 if (m_chatHistoryCache.contains(chatPartner)) {
973 m_chatHistoryCache[chatPartner].messages.append(incomingMsg);
974 }
975
976 // 4. Отправляем на сервер подтверждение о ДОСТАВКЕ.
977 // Это говорит серверу, что клиент получил сообщение.
978 QJsonObject deliveredCmd;
979 deliveredCmd["type"] = "message_delivered";
980 deliveredCmd["id"] = (double)incomingMsg.id;
981 sendJson(deliveredCmd);
982
983 // 5. Логика отображения в зависимости от того, какой чат сейчас открыт.
984 if (incomingMsg.fromUser == m_currentChatPartner.username) {
985 // --- Сообщение пришло для ТЕКУЩЕГО открытого чата ---
986 bool wasScrolledToBottom = m_chatViewWidget->isScrolledToBottom();
987
988 // Добавляем сообщение в `ChatMessageModel`, что приводит к его отображению.
989 m_chatModel->addMessage(incomingMsg);
990
991 if (wasScrolledToBottom) {
992 // Если пользователь и так был внизу, плавно прокручиваем к новому сообщению.
993 QMetaObject::invokeMethod(m_chatViewWidget, "scrollToBottom", Qt::QueuedConnection);
994 } else {
995 // Если пользователь читал историю, эмитируем сигнал для показа кнопки "вниз".
996 emit newMessageForCurrentChat();
997 }
998
999 // С небольшой задержкой запускаем проверку видимых сообщений,
1000 // чтобы отправить статус "прочитано" для этого нового сообщения.
1001 QTimer::singleShot(50, this, &MainWindow::processVisibleMessages);
1002 } else {
1003 // --- Сообщение пришло для ДРУГОГО (неактивного) чата ---
1004 m_unreadCounts[incomingMsg.fromUser]++; // Увеличиваем счетчик непрочитанных.
1005 m_contactModel->refreshContact(incomingMsg.fromUser); // Просим модель перерисовать этот контакт.
1006 }
1007
1008 // 6. Показываем системное уведомление (мигание иконки), если окно неактивно
1009 // или сообщение пришло для другого чата.
1010 if (incomingMsg.fromUser != m_currentChatPartner.username || isMinimized() || !isActiveWindow()) {
1011 QApplication::alert(this);
1012 }
1013}

1188void MainWindow::updateMessageStatusInCacheAndModel(qint64 messageId, ChatMessage::MessageStatus newStatus)
1189{
1190 // Проходимся по всему кэшу чатов.
1191 for (auto it = m_chatHistoryCache.begin(); it != m_chatHistoryCache.end(); ++it) {
1192 QList<ChatMessage>& messages = it.value().messages;
1193 // Проходимся по всем сообщениям в одном чате.
1194 for (int i = 0; i < messages.size(); ++i) {
1195 if (messages[i].id == messageId) {
1196 // Нашли нужное сообщение.
1197 if (messages[i].status == newStatus) return; // Статус уже такой, ничего не делаем.
1198
1199 // 1. Обновляем статус в "источнике правды" - кэше.
1200 messages[i].status = newStatus;
1201 QString foundInChatWith = it.key();
1202 qDebug() << "[CACHE] Status updated for message" << messageId << "in chat with" << foundInChatWith;
1203
1204 // 2. Если этот чат сейчас открыт, обновляем и активную модель.
1205 if (foundInChatWith == m_currentChatPartner.username) {
1206 // Используем setData для эффективного обновления, а не `updateMessageStatus`.
1207 // Это заставит view перерисовать только один элемент.
1208 QModelIndex modelIndex = m_chatModel->index(i, 0);
1209 m_chatModel->setData(modelIndex, QVariant::fromValue(messages[i]), Qt::UserRole);
1210 }
1211
1212 return; // Сообщение найдено и обработано, выходим из всех циклов.
1213 }
1214 }
1215 }

1884void MainWindow::handleOldHistoryData(const QJsonObject& response)
1885{
1886 // 1. Проверяем, что история пришла для текущего открытого чата.
1887 QString historyForUser = response["with_user"].toString();
1888 if (historyForUser != m_currentChatPartner.username) {
1889 m_isLoadingHistory = false; // Сбрасываем флаг и выходим.
1890 return;
1891 }
1892
1893 QJsonArray history = response["history"].toArray();
1894 // Если сервер вернул пустой массив, это значит, что мы достигли начала
1895 // всей истории переписки.
1896 if (history.isEmpty()) {
1897 m_oldestMessageId = 0; // 0 - это наш флаг, означающий "больше загружать нечего".
1898 m_isLoadingHistory = false;
1899 m_chatHistoryCache[historyForUser].allMessagesLoaded = true; // Отмечаем в кэше.
1900 return;
1901 }
1902 qDebug() << "[CLIENT] Prepending" << history.count() << "older messages for" << historyForUser;
1903
1904 // --- 2. Получаем текущее состояние скроллбара ДО изменений ---
1905 QListView* chatView = m_chatViewWidget->chatHistoryView();
1906 QScrollBar* scrollBar = chatView->verticalScrollBar();
1907 // Запоминаем максимальное значение высоты контента.
1908 int oldScrollMax = scrollBar->maximum();
1909
1910 // --- 3. Парсинг и обработка сообщений ---
1911 // (Этот блок идентичен `handleHistoryData`)
1912 QList<ChatMessage> messages;
1913 for (int i = 0; i < history.count(); ++i) {
1914 const QJsonValue &value = history[i];
1915 ChatMessage msg;
1916 QJsonObject msgObj = value.toObject();
1917 msg.id = msgObj["id"].toDouble();
1918 msg.fromUser = msgObj["fromUser"].toString();
1919 msg.toUser = msgObj["toUser"].toString();
1920 msg.payload = msgObj["payload"].toString();
1921 msg.timestamp = msgObj["timestamp"].toString();
1922 msg.replyToId = msgObj["reply_to_id"].toDouble();
1923 msg.isOutgoing = (msg.fromUser == m_currentUsername);
1924 msg.isEdited = msgObj["is_edited"].toInt();
1925 if(msgObj["is_delivered"].toInt() == 1){
1926 msg.status = ChatMessage::Delivered;
1927 }
1928 else{
1929 msg.status = ChatMessage::Sent;
1930 }
1931 if(msgObj["is_read"].toInt() == 1){
1932 msg.status = ChatMessage::Read;
1933 }
1934 messages.append(msg);
1935 }
1936
1937 // --- 4. Обновление кэша и модели ---
1938 ChatCache& cache = m_chatHistoryCache[historyForUser];
1939
1940 // Добавляем новые (старые) сообщения в НАЧАЛО списка в кэше.
1941 for (int i = messages.count() - 1; i >= 0; --i) {
1942 cache.messages.prepend(messages.at(i));
1943 }

1957
1958 // `QApplication::processEvents()` - это небольшой "хак". Мы просим Qt
1959 // немедленно обработать все отложенные события, включая изменения в модели
1960 // и перерисовку QListView. Это гарантирует, что скроллбар обновит свое
1961 // максимальное значение (`maximum`) ДО того, как мы попытаемся его установить.
1962 QApplication::processEvents();
1963

1965 int newScrollMax = scrollBar->maximum();
1966
1967 // Устанавливаем новую позицию. Разница `newScrollMax - oldScrollMax` - это
1968 // в точности высота добавленных сверху сообщений. Таким образом, мы "проталкиваем"
1969 // скроллбар вниз на высоту нового контента, и для пользователя видимая область
1970 // остается на том же самом сообщении, на котором он был до подгрузки.
1971 scrollBar->setValue(newScrollMax - oldScrollMax);
1972
1973 QTimer::singleShot(50, this, &MainWindow::processVisibleMessages);
1974
1975 // Сбрасываем флаг, разрешая следующие подгрузки.
1976 m_isLoadingHistory = false;
