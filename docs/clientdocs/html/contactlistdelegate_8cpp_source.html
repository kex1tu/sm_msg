35
36 // --- 1. Отрисовка фона ---
37 // Вручную рисуем фон для состояний "выделен" и "наведен", чтобы иметь полный контроль.
38 if (option.state & QStyle::State_Selected) {
39 painter->fillRect(option.rect, option.palette.highlight()); // Цвет выделения из палитры.
40 } else if (option.state & QStyle::State_MouseOver) {
41 painter->fillRect(option.rect, option.palette.color(QPalette::AlternateBase)); // Альтернативный цвет фона.
42 }
43
44 // --- 2. Получение данных из модели ---
45 // Запрашиваем всю необходимую информацию у модели по кастомным ролям.
46 QString displayName = index.data(Qt::DisplayRole).toString();
47 QString lastMessage = index.data(ContactListModel::LastMessageRole).toString();
48 bool isOnline = index.data(ContactListModel::IsOnlineRole).toBool();
49 bool isTyping = index.data(ContactListModel::IsTypingRole).toBool();
50 int unreadCount = index.data(ContactListModel::UnreadCountRole).toInt();
51
52 // --- 3. Подготовка текста для нижней строки ---
53 QString bottomText = lastMessage;
54 QColor bottomTextColor = Qt::gray;
55
56 // Статус "печатает..." имеет наивысший приоритет.
57 if (isTyping) {
58 bottomText = "печатает...";
59 bottomTextColor = QColor("#F4ABC4"); // Акцентный цвет.
60 }
61
62 // --- 4. Расчет геометрии для текста ---
63 int padding = 12; // Боковые отступы.
64 QRect contentRect = option.rect.adjusted(padding, padding / 2, -padding, -padding / 2);
65 // Делим доступное пространство на два прямоугольника: для верхней и нижней строки.
66 QRect topRect(contentRect.left(), contentRect.top(), contentRect.width(), contentRect.height() / 2);
67 QRect bottomRect(contentRect.left(), contentRect.top() + contentRect.height() / 2, contentRect.width(), contentRect.height() / 2);
68
69 // --- 5. Отрисовка верхней строки (Имя пользователя) ---
70 QFont nameFont = painter->font();
71 nameFont.setBold(isOnline); // Жирный шрифт для онлайн-пользователей.
72 painter->setFont(nameFont);
73 painter->setPen(isOnline ? Qt::white : QColor("#EAEAEA"));
74 painter->drawText(topRect, Qt::AlignLeft | Qt::AlignBottom, displayName);
75
76 // --- 6. Отрисовка нижней строки (Превью сообщения или статус) ---
77 QFontMetrics fm(painter->font());
78 // Обрезаем длинный текст и добавляем многоточие.
79 QString elidedBottomText = fm.elidedText(bottomText, Qt::ElideRight, bottomRect.width());
80 painter->setFont(QApplication::font()); // Возвращаем стандартный шрифт.
81 painter->setPen(bottomTextColor);
82 painter->drawText(bottomRect, Qt::AlignLeft | Qt::AlignTop, elidedBottomText);
83
84 // --- 7. Отрисовка "бейджа" с количеством непрочитанных сообщений ---
85 if (unreadCount > 0) {
86 painter->setRenderHint(QPainter::Antialiasing); // Включаем сглаживание для круга.
87
88 int badgeSize = 20; // Диаметр кружка.
89 QColor badgeColor = option.palette.highlight().color(); // Используем цвет выделения для фона.

91
92 // Рассчитываем положение "бейджа" в правом углу элемента.
93 QRect badgeRect(
94 option.rect.right() - badgeSize - padding,
95 option.rect.top() + (option.rect.height() - badgeSize) / 2,
96 badgeSize, badgeSize
97 );
98

100 painter->setBrush(badgeColor);
101 painter->setPen(Qt::NoPen);
102 painter->drawEllipse(badgeRect);
103
104 // Рисуем текст (цифру).
105 painter->setPen(textColor);
106 painter->setFont(QFont("Segoe UI", 8, QFont::Bold));
107 painter->drawText(badgeRect, Qt::AlignCenter, QString::number(unreadCount));
