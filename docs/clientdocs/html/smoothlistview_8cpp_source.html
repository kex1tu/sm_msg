19SmoothListView::SmoothListView(QWidget *parent)
20 : QListView(parent)
21{
22 // 1. Устанавливаем режим попиксельной прокрутки.
23 // Вместо стандартного режима "по элементам" (ScrollPerItem), этот режим
24 // позволяет плавно сдвигать содержимое на любое количество пикселей.
25 // Это основа для плавной прокрутки.
26 setVerticalScrollMode(QAbstractItemView::ScrollPerPixel);
27
28 // 2. Отключаем горизонтальный скроллбар, так как наш контент (сообщения)
29 // должен всегда подстраиваться под ширину виджета.
30 setHorizontalScrollBarPolicy(Qt::ScrollBarAlwaysOff);
31
32 // 3. Устанавливаем режим пакетной компоновки (Batched layout).
33 // Это оптимизация производительности. Вместо того чтобы пересчитывать
34 // геометрию всех элементов при каждом небольшом изменении, QListView
35 // будет делать это "пачками" по `batchSize` элементов, что снижает
36 // нагрузку при работе с очень большими списками.
37 setLayoutMode(QListView::Batched);
38 setBatchSize(100); // Размер "пачки".
39}

45void SmoothListView::wheelEvent(QWheelEvent *e)
46{
47 // Стандартная прокрутка в режиме ScrollPerPixel может быть слишком быстрой.
48 // Мы перехватываем событие, чтобы уменьшить его интенсивность.
49
50 // 1. Получаем "дельта-угол" — насколько было повернуто колесо.
51 QPoint angleDelta = e->angleDelta();
52
53 // 2. Определяем коэффициент замедления. 0.5 означает, что прокрутка
54 // будет в два раза медленнее стандартной.
55 double scrollFactor = 0.5;
56
57 // 3. Создаем НОВОЕ событие QWheelEvent, копируя все поля из старого,
58 // но умножая `angleDelta` на наш коэффициент.
59 QWheelEvent newEvent(
60 e->position(),
61 e->globalPosition(),
62 e->pixelDelta(),
63 QPoint(angleDelta.x() * scrollFactor, angleDelta.y() * scrollFactor),
64 e->buttons(),
65 e->modifiers(),
66 e->phase(),
67 e->inverted()
68 );
69
70 // 4. Передаем наше измененное, "ослабленное" событие в стандартный
71 // обработчик базового класса QListView.
72 QListView::wheelEvent(&newEvent);
73}

79void SmoothListView::resizeEvent(QResizeEvent *e)
80{
81 // 1. Сначала вызываем стандартный обработчик.
82 QListView::resizeEvent(e);
83
84 // 2. Затем принудительно вызываем `doItemsLayout()`.
85 // Это необходимо, потому что в режиме пакетной компоновки и попиксельной
86 // прокрутки QListView может не сразу обновить расположение элементов
87 // при изменении ширины, что приводит к визуальным артефактам.
88 // Этот вызов заставляет его немедленно пересчитать геометрию всех видимых элементов.
89 doItemsLayout();
90}

101bool SmoothListView::viewportEvent(QEvent *event)
102{
103 switch (event->type()) {
104 // Если мышь вошла в область виджета, двигается над ним или покинула его...
105 case QEvent::HoverEnter:
106 case QEvent::HoverMove:
107 case QEvent::HoverLeave:
108 // ... мы "принимаем" событие. Это гарантирует, что Qt продолжит
109 // отслеживать положение мыши и отправлять события наведения,
110 // которые нужны для hover-эффектов в делегате.
111 event->accept();
112 return true;
113 default:
114 break;
115 }
116
117 // Для всех остальных типов событий вызываем стандартный обработчик.
118 return QListView::viewportEvent(event);
119}
